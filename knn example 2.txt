import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import accuracy_score

# Step 1: Create a small dataset
# Features: [Feature1, Feature2]
data = {
    'Feature1': [1, 2, 3, 6, 7, 8],
    'Feature2': [1, 1, 2, 5, 5, 6],
    'Target': [0, 0, 0, 1, 1, 1]  # Labels: 0 for Class 0, 1 for Class 1
}

df = pd.DataFrame(data)

# Step 2: Prepare the features and labels
X = df[['Feature1', 'Feature2']].values  # Features
y = df['Target'].values  # Labels

# Step 3: Split the dataset into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.33, random_state=42)

# Step 4: Initialize and train the KNN classifier
k = 3  # Number of neighbors
knn = KNeighborsClassifier(n_neighbors=k)

# Fit the model
knn.fit(X_train, y_train)

# Step 5: Make predictions and calculate accuracy
y_pred = knn.predict(X_test)
accuracy = accuracy_score(y_test, y_pred)

print(f"Accuracy: {accuracy * 100:.2f}%")

# Example prediction for a new data point
new_data = np.array([[5, 4]])  # Example point
prediction = knn.predict(new_data)
print(f"Predicted class for {new_data[0]}: {prediction[0]}")
